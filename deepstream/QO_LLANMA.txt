ISHGA TUSHIRISH (Ubuntu Desktop'da)
────────────────────────────────────────────────────────────────────
  # 1. Modelni yuklash (bir martalik)
  bash deepstream/scripts/download_models.sh   # yoki qo'lda

  # 2. Video qo'yish
  cp /path/to/sinf_video.mp4 deepstream/data/sinf.mp4

  # 3. Camera ID — qaysi kameraga bog'lash
  echo "DEEPSTREAM_CAMERA_ID=1" >> .env

  # 4. Ishga tushirish (alohida profil)
  docker compose --profile deepstream up -d

  # 5. Loglar
  docker compose logs -f deepstream         # pipeline
  docker compose logs -f kafka_consumer     # davomat

  # 6. Kafka UI
  # http://localhost:8080
